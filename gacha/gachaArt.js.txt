// gacha-art.js — Loofta Gacha Pack artwork.
// Classic white/black/gold gacha booster layout, Loofta brand, pokéball medallion.
// Single source of truth: the preview page and the React component both draw from here.

export const AW = 760;
export const AH = 1120;

const RED = '#FF1301';
const ORG = '#FF6A00';
const YEL = '#EAB308';
const GOLD_L = '#FFE9A8';
const GOLD_M = '#E8B23A';
const GOLD_D = '#9A6B12';
const INK = '#0B0B0F';
const PAPER = '#FFFDF7';

export function loadSvg(url) {
  return fetch(url).then(r => r.text()).then(t => {
    const uri = 'data:image/svg+xml;base64,' + btoa(unescape(encodeURIComponent(t)));
    return new Promise(res => { const i = new Image(); i.onload = () => res(i); i.onerror = () => res(null); i.src = uri; });
  }).catch(() => null);
}

function goldGrad(x, x0, y0, x1, y1) {
  const g = x.createLinearGradient(x0, y0, x1, y1);
  g.addColorStop(0, GOLD_D); g.addColorStop(0.22, GOLD_M); g.addColorStop(0.5, GOLD_L);
  g.addColorStop(0.78, GOLD_M); g.addColorStop(1, GOLD_D);
  return g;
}
function brandGrad(x, x0, y0, x1, y1) {
  const g = x.createLinearGradient(x0, y0, x1, y1);
  g.addColorStop(0, RED); g.addColorStop(0.55, ORG); g.addColorStop(1, YEL);
  return g;
}
function star(x, cx, cy, r) {
  x.beginPath();
  for (let i = 0; i < 10; i++) {
    const rad = i % 2 ? r * 0.44 : r;
    const a = -Math.PI / 2 + (i * Math.PI) / 5;
    const px = cx + Math.cos(a) * rad, py = cy + Math.sin(a) * rad;
    i ? x.lineTo(px, py) : x.moveTo(px, py);
  }
  x.closePath();
}
function crimp(x, y, h, dir) {
  x.save();
  x.fillStyle = INK; x.globalAlpha = 0.9; x.fillRect(0, y, AW, h);
  x.globalAlpha = 0.16; x.fillStyle = '#fff';
  for (let i = 0; i < AW; i += 9) x.fillRect(i, y, 3, h);
  x.globalAlpha = 0.9; x.fillStyle = INK;
  const teeth = 30, tw = AW / teeth, base = dir > 0 ? y + h : y, tip = base + dir * 14;
  x.beginPath(); x.moveTo(0, base);
  for (let i = 0; i < teeth; i++) { x.lineTo(i * tw + tw / 2, tip); x.lineTo((i + 1) * tw, base); }
  x.closePath(); x.fill();
  x.restore();
}

/** Pokéball-style medallion: brand-gradient dome, cream base, black band, mystery button. */
function medallion(x, cx, cy, r) {
  x.save();
  // outer gold bezel
  x.beginPath(); x.arc(cx, cy, r, 0, 7);
  x.fillStyle = goldGrad(x, cx - r, cy - r, cx + r, cy + r); x.fill();
  x.lineWidth = 3; x.strokeStyle = GOLD_D; x.stroke();

  const ri = r - 26;
  // upper dome
  x.save();
  x.beginPath(); x.arc(cx, cy, ri, Math.PI, 0); x.closePath(); x.clip();
  x.fillStyle = brandGrad(x, cx - ri, cy - ri, cx + ri, cy);
  x.fillRect(cx - ri, cy - ri, ri * 2, ri);
  x.globalAlpha = 0.28; x.fillStyle = '#fff';
  x.beginPath(); x.ellipse(cx - ri * 0.3, cy - ri * 0.55, ri * 0.42, ri * 0.22, -0.5, 0, 7); x.fill();
  x.restore();
  // lower half
  x.save();
  x.beginPath(); x.arc(cx, cy, ri, 0, Math.PI); x.closePath(); x.clip();
  const lg = x.createLinearGradient(0, cy, 0, cy + ri);
  lg.addColorStop(0, '#FFFFFF'); lg.addColorStop(1, '#E4DFD3');
  x.fillStyle = lg; x.fillRect(cx - ri, cy, ri * 2, ri);
  x.restore();
  // equator band
  x.save();
  x.beginPath(); x.arc(cx, cy, ri, 0, 7); x.clip();
  x.fillStyle = INK; x.fillRect(cx - ri, cy - ri * 0.135, ri * 2, ri * 0.27);
  x.fillStyle = goldGrad(x, cx - ri, 0, cx + ri, 0);
  x.fillRect(cx - ri, cy - ri * 0.152, ri * 2, 4);
  x.fillRect(cx - ri, cy + ri * 0.135, ri * 2, 4);
  x.restore();
  // mystery button
  const br = r * 0.3;
  x.beginPath(); x.arc(cx, cy, br + 12, 0, 7);
  x.fillStyle = goldGrad(x, cx - br, cy - br, cx + br, cy + br); x.fill();
  x.beginPath(); x.arc(cx, cy, br, 0, 7); x.fillStyle = PAPER; x.fill();
  x.lineWidth = 3; x.strokeStyle = GOLD_D; x.stroke();
  x.textAlign = 'center';
  x.fillStyle = brandGrad(x, cx - br, cy - br, cx + br, cy + br);
  x.font = `500 ${Math.round(br * 1.5)}px "Clash Display"`;
  x.fillText('?', cx, cy + br * 0.53);
  x.restore();
}

/**
 * Draw the pack onto a 2D context sized AW×AH.
 * imgs: { logoWhite, logoDark, megapotDark, megapotWhite } — HTMLImageElements or null.
 */
export function drawGachaPack(x, face = 'front', imgs = {}) {
  const { logoWhite, logoDark, megapotDark, megapotWhite } = imgs;
  x.clearRect(0, 0, AW, AH);

  // pearl-white foil body
  const base = x.createLinearGradient(0, 0, AW, AH);
  base.addColorStop(0, '#FFFFFF'); base.addColorStop(0.44, '#F4F1E9');
  base.addColorStop(0.7, '#FFFFFF'); base.addColorStop(1, '#E9E5DA');
  x.fillStyle = base; x.fillRect(0, 0, AW, AH);
  // diagonal sheen
  x.save(); x.translate(AW / 2, AH / 2); x.rotate(-0.62); x.globalAlpha = 0.5;
  const sh = x.createLinearGradient(-AW, 0, AW, 0);
  sh.addColorStop(0, 'rgba(255,255,255,0)'); sh.addColorStop(0.5, 'rgba(255,255,255,.85)');
  sh.addColorStop(1, 'rgba(255,255,255,0)');
  x.fillStyle = sh; x.fillRect(-AW, -170, AW * 2, 340); x.restore();

  // ---- black sweeps ----
  x.fillStyle = INK;
  x.beginPath();
  x.moveTo(0, 0); x.lineTo(AW, 0); x.lineTo(AW, 132);
  x.bezierCurveTo(AW * 0.64, 322, AW * 0.30, 292, 0, 196);
  x.closePath(); x.fill();
  x.beginPath();
  x.moveTo(0, AH); x.lineTo(AW, AH); x.lineTo(AW, AH - 248);
  x.bezierCurveTo(AW * 0.60, AH - 430, AW * 0.26, AH - 392, 0, AH - 286);
  x.closePath(); x.fill();

  // gold hairlines tracing the sweeps
  x.lineWidth = 6; x.strokeStyle = goldGrad(x, 0, 0, AW, 0);
  x.beginPath(); x.moveTo(AW, 152);
  x.bezierCurveTo(AW * 0.64, 342, AW * 0.30, 312, 0, 216); x.stroke();
  x.beginPath(); x.moveTo(AW, AH - 268);
  x.bezierCurveTo(AW * 0.60, AH - 450, AW * 0.26, AH - 412, 0, AH - 306); x.stroke();

  // arcs framing the medallion
  const cx = AW / 2, cy = face === 'front' ? 556 : 470, r = face === 'front' ? 186 : 120;
  x.lineWidth = 5; x.strokeStyle = goldGrad(x, cx - 330, 0, cx + 330, 0);
  x.beginPath(); x.arc(cx, cy, r + 92, 2.55, 3.95); x.stroke();
  x.beginPath(); x.arc(cx, cy, r + 92, -0.6, 0.8); x.stroke();
  x.globalAlpha = 0.5;
  x.beginPath(); x.arc(cx, cy, r + 118, 2.7, 3.8); x.stroke();
  x.beginPath(); x.arc(cx, cy, r + 118, -0.45, 0.65); x.stroke();
  x.globalAlpha = 1;

  // ---- brand marks ----
  x.textAlign = 'center';
  if (logoWhite) {
    const w = 300, h = w * logoWhite.height / logoWhite.width;
    x.drawImage(logoWhite, (AW - w) / 2, 126, w, h);
  }

  medallion(x, cx, cy, r);

  if (face === 'front') {
    // five gold stars
    const sy = 826, gap = 96;
    for (let i = 0; i < 5; i++) {
      const sx = AW / 2 + (i - 2) * gap;
      x.save();
      x.shadowColor = 'rgba(154,107,18,.45)'; x.shadowBlur = 10; x.shadowOffsetY = 3;
      star(x, sx, sy, 34);
      x.fillStyle = goldGrad(x, sx - 34, sy - 34, sx + 34, sy + 34); x.fill();
      x.restore();
      x.lineWidth = 2; x.strokeStyle = GOLD_D; star(x, sx, sy, 34); x.stroke();
    }
    // bottom lockup: GACHA | PACK
    const ly = AH - 128;
    x.font = '500 62px "Clash Display"';
    const gw = x.measureText('GACHA').width;
    const pad = 22, bw = gw + pad * 2, bx = AW / 2 - bw - 16;
    x.beginPath();
    if (x.roundRect) x.roundRect(bx, ly - 52, bw, 74, 10); else x.rect(bx, ly - 52, bw, 74);
    x.fillStyle = goldGrad(x, bx, ly - 52, bx + bw, ly + 22); x.fill();
    x.fillStyle = INK; x.textAlign = 'center';
    x.fillText('GACHA', bx + bw / 2, ly);
    x.textAlign = 'left';
    x.fillStyle = '#fff'; x.fillText('PACK', AW / 2 + 4, ly);
    // divider
    x.fillStyle = goldGrad(x, 0, 0, 0, AH);
    x.fillRect(AW / 2 - 10, ly - 46, 3, 62);
  } else {
    // back: brand mark + medallion only
  }

  // sealed edges
  crimp(x, 0, 46, 1);
  crimp(x, AH - 46, 46, -1);
  // hang hole
  x.save(); x.globalAlpha = 0.55; x.fillStyle = '#3A3A44';
  x.beginPath(); x.ellipse(AW / 2, 74, 30, 13, 0, 0, 7); x.fill();
  x.restore();
}
